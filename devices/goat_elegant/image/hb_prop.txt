# Lumen OS on the XGIMI Z9X HB (goat_elegant, G0130), FIRST-BOOT TEST builds only (never a release). Appended to
# /system/product/etc/build.prop by compose_hb.sh; init reads that file last, so these lines win over the base and
# the vendor. compose_hb.sh adds ro.lumen.version, ro.lumen.build_id and, with --usb-adb, persist.sys.usb.config=adb.
# Comment lines are dropped when appended.
ro.lumen.device=goat_elegant
# adb without the key prompt (the HB vendor build.prop already has ro.adb.secure=0; kept here so the image does not
# depend on it) and adb on TCP 5555 as stock runs it (getprop.txt service.adb.tcp.port=5555)
ro.adb.secure=0
service.adb.tcp.port=5555
# GPU driver name: the vendor ships only /vendor/lib/egl/libGLES_mali.so and sets no ro.hardware.egl; Android 14
# tries libGLES_<ro.hardware.egl> and libGLES_<ro.board.platform> (mt5873) only (Play 6 t5: surfaceflinger aborted
# with "couldn't find an OpenGL ES implementation" until this line)
ro.hardware.egl=mali
# the base product build.prop says "emulator"; stock and every Lumen image say tv
ro.build.characteristics=tv
# the vendor sets ro.product.locale=zh-CN (a China build); English as on the Z9X and Play 6 images
ro.product.locale=en-US
# test builds: logcat without the SELinux audit flood (still in dmesg) and 2 MB buffers, so the boot's own errors
# stay in the USB-stick log (the Play 6 --diag lines)
ro.logd.auditd=false
persist.logd.size=2097152
