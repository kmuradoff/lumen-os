# dump), and never vbmeta (the user's one-time step, never written by an update).

# the wire id: the manifest's "device" and the payload's pre-device must be this (LumenUpdater Ota.device())
ro.lumen.ota.device=hb
# the update manifest on the fixed pre-release tag z9x-hb, the HB's update channel (never releases/latest, which is
